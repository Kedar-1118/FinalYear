\section{Theoretical Analysis and PAC Guarantees}
\label{sec:theoretical_guarantees}

This section establishes the formal theoretical guarantees of MABSplit, proving global $(\epsilon, \delta)$-PAC optimality and deriving its finite-population sample complexity bounds.

\subsection{Error Budget Conservation across Quantization Tiers}
\begin{lemma}[Bonferroni Tier Budget Conservation]
\label{lem:bonferroni}
Let $\mathcal{E}_{\text{coarse}}$ represent the failure event that the true optimal split arm $a^*$ is inadvertently discarded during Phase 1 with confidence parameter $\delta_{\text{coarse}} = \delta/3$. Let $\mathcal{E}_{\text{fine}}$ be the event that an $\epsilon$-suboptimal arm $\hat{a}$ ($\mu_{a^*} - \mu_{\hat{a}} > \epsilon$) survives Phase 2 under confidence parameter $\delta_{\text{fine}} = 2\delta/3$. The composite error probability across both tiers satisfies:
\begin{equation}
\mathbb{P}(\mathcal{E}_{\text{total}}) = \mathbb{P}(\mathcal{E}_{\text{coarse}} \cup \mathcal{E}_{\text{fine}}) \le \delta
\end{equation}
\end{lemma}

\begin{proof}
Applying Boole's inequality (union bound) across sequential dependent phases:
\begin{align}
\mathbb{P}(\mathcal{E}_{\text{total}}) &\le \mathbb{P}(\mathcal{E}_{\text{coarse}}) + \mathbb{P}(\mathcal{E}_{\text{fine}} \mid \neg\mathcal{E}_{\text{coarse}}) \\
&\le \delta_{\text{coarse}} + \delta_{\text{fine}} = \frac{\delta}{3} + \frac{2\delta}{3} = \delta
\end{align}
Thus, segmenting the error quota into a one-third fraction for coarse pruning and a two-thirds fraction for fine resolution guarantees that the overall family-wise error rate remains strictly capped at $\delta$.
\end{proof}

\subsection{$(\epsilon, \delta)$-PAC Optimality Guarantees}
\begin{theorem}[PAC Optimality of MABSplit]
\label{thm:pac_optimality}
For any partition dataset $\mathcal{D}$ with $N$ instances, user-defined failure tolerance $\delta \in (0, 1)$, and sub-optimality allowance $\epsilon \ge 0$, Algorithm~\ref{alg:mabsplit} outputs an identified split arm $\hat{a}$ satisfying:
\begin{equation}
\mathbb{P}\left(\Delta I(a^*, \mathcal{D}) - \Delta I(\hat{a}, \mathcal{D}) \le \epsilon \right) \ge 1 - \delta
\end{equation}
\end{theorem}

\begin{proof}
By Serfling's finite-population concentration inequality \cite{serfling1974probability,bardenet2015concentration}, for any candidate arm $a$ evaluated across $n$ non-replacement minibatch draws from node cohort $N$:
\begin{equation}
\mathbb{P}\left(|\hat{\mu}_a(n) - \mu_a| \ge \text{Rad}_{\text{Serfling}}(n, N, \delta') \right) \le \delta'
\end{equation}
When arm $a$ is pruned at iteration $t$, its upper statistical bound lies strictly below the empirical lower bound of some competing arm $a'$, assuring $\mu_{a^*} - \mu_a > 0$ with probability at least $1 - \delta'$. Upon algorithm termination, the surviving arm $\hat{a}$ satisfies $\mu_{a^*} - \mu_{\hat{a}} \le \epsilon$. By Lemma~\ref{lem:bonferroni}, union-bounding over Phase 1 and Phase 2 verifies that the global execution error does not exceed $\delta$.
\end{proof}

\subsection{Finite-Population Sample Complexity}
\begin{corollary}[Asymptotic Query Complexity]
Let $\Delta_a = \mu_{a^*} - \mu_a$ represent the sub-optimality margin for suboptimal candidate $a$. The cumulative sample queries evaluated by MABSplit satisfy:
\begin{equation}
\mathcal{T}_{\text{MABSplit}} = \mathcal{O}\left( \sum_{a \neq a^*} \min\left( N, \frac{\sigma_a^2 \ln(K/\delta)}{\Delta_a^2} + \frac{\ln(K/\delta)}{\Delta_a} \right) \right)
\end{equation}
\end{corollary}
\noindent\textbf{Analytical Implication:} Under classical i.i.d. Hoeffding bounds, sample evaluations on competitive arms can theoretically exceed $N$ due to conservative confidence widths. In contrast, the finite-population coefficient $\kappa(n, N) = 1 - \frac{n-1}{N}$ naturally truncates individual arm queries at $N$, ensuring the algorithm smoothly transitions to exact ground truth without artificial sample inflation.
